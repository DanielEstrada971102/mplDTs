Example

super_layer = SuperLayer(rawId=589357056)
print(super_layer)
for layer in super_layer.layers:
    print("\t", layer)
    print(2 * "\t", layer.cell(layer._first_cell_id))
    print(2 * "\t", layer.cell(len(layer.cells) - 1))
